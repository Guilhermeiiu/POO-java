package Sistema3;
import java.util.Date;

public class Main {
    public static void main(String[] args) {
        // criando produtos audiovisuais
        Filme filme = new Filme(101L, "Oppenheimer", 2023, "Christopher Nolan", 180, true);
        filme.adicionarAtor("Cillian Murphy");
        filme.adicionarAtor("Emily Blunt");
        filme.adicionarAtor("Robert Downey Jr.");

        Serie serie = new Serie(102L, "Severance", 2022, "Ben Stiller", 2);
        serie.adicionarAtor("Adam Scott");
        serie.adicionarAtor("Britt Lower");

        Documentario doc = new Documentario(103L, "Free Solo", 2018, "Elizabeth Chai Vasarhelyi", 100, true);
        
        // criando avaliações
        Avaliacao av1 = new Avaliacao(1L, new Date(), filme, "Obra-prima! Fotografia e trilha sonora impecáveis.");
        Avaliacao av2 = new Avaliacao(2L, new Date(), serie, "Roteiro envolvente e final de temporada tenso.");
        Avaliacao av3 = new Avaliacao(3L, new Date(), doc, "De tirar o fôlego do início ao fim.");

        // criando usuário e vinculando avaliações
        Usuario usuario = new Usuario(1L, "Guilherme");
        usuario.adicionarAvaliacao(av1);
        usuario.adicionarAvaliacao(av2);
        usuario.adicionarAvaliacao(av3);

        // exibindo todas as avaliações do usuário
        usuario.imprimirAvaliacoes();

        // testando alterações e remoção
        av2.alterarAvaliacao("Melhor série de ficção científica recente! Nota 10.");
        usuario.removerAvaliacao(av3);

        System.out.println("--- APÓS ATUALIZAÇÃO ---\n");
        usuario.imprimirAvaliacoes();
    }
}
package Sistema3;

import java.util.HashSet;
import java.util.Set;

public class Usuario {
    private long id;
    private String nome;
    private Set<Avaliacao> avaliacoes;

    public Usuario(long id, String nome) {
        this.id = id;
        this.nome = nome;
        this.avaliacoes = new HashSet<>();
    }

    public void alterarNome(String nome) {
        this.nome = nome;
    }

    public void adicionarAvaliacao(Avaliacao av) {
        avaliacoes.add(av);
    }

    public void removerAvaliacao(Avaliacao av) {
        avaliacoes.remove(av);
    }

    public void imprimirAvaliacoes() {
        System.out.println("Avaliações de " + nome + " (ID: " + id + ")\n");
        if (avaliacoes.isEmpty()) {
            System.out.println("Nenhuma avaliação registrada.");
            return;
        }
        for (Avaliacao av : avaliacoes) {
            av.imprimirAvaliacao();
        }
    }
}
package Sistema3;

public class Documentario extends ProdutoAudioVisual {
    private boolean ganhouOscar;
    private int duracao;

    public Documentario(long id, String titulo, int ano, String diretor, int duracao, boolean ganhouOscar) {
        super(id, titulo, ano, diretor);
        this.ganhouOscar = ganhouOscar;
        this.duracao = duracao;
    }

    public void ganhouOscar(boolean ganhou) {
        this.ganhouOscar = ganhou;
    }

    public void alterarDuracao(int minutos) {
        this.duracao = minutos;
    }

    @Override
    public void imprimir() {
        super.imprimir();
        System.out.println("Duração: " + duracao + " min");
        System.out.println(ganhouOscar ? "Ganhou o Oscar!!" : "Não ganhou o Oscar");
    }
}
package Sistema3;

public abstract class ProdutoAudioVisual {
    protected long id;
    protected String titulo;
    protected int ano;
    protected String diretor;

    public ProdutoAudioVisual(long id, String titulo, int ano, String diretor) {
        this.id = id;
        this.titulo = titulo;
        this.ano = ano;
        this.diretor = diretor;
    }

    public void alterarTitulo(String titulo) {
        this.titulo = titulo;
    }

    public void alterarAno(int ano) {
        this.ano = ano;
    }
    public void alterarDiretor(String diretor) {
        this.diretor = diretor;
    }
    public void imprimir() {
        System.out.println("ID: " + id + " | Título: " + titulo + "\nAno: " + ano + " | Diretor: " + diretor);
    }
}
package Sistema3;
import java.util.*;
public class Serie extends ProdutoAudioVisual{
	Set <String> atores;
	int temporadas;
	Serie(long id,String titulo, int ano, String diretor,int temporadas) {
		super(id, titulo, ano, diretor);
		this.atores=new HashSet<>();
		this.temporadas=temporadas;
	}
	public void adicionarAtor(String nome) {
		atores.add(nome);
	}
	public void removerAtor(String nome) {
		atores.remove(nome);
	}
	public void alterarNumTemporadas(int temporadas) {
		this.temporadas=temporadas;
	}
	@Override
	public void imprimir() {
		super.imprimir();
		System.out.println("Atores: "+atores);
		System.out.println("\nNúmero de temporadas: "+temporadas);
	}

}
package Sistema3;

import java.util.Date;

public class Avaliacao {
    private long id;
    private Date data;
    private ProdutoAudioVisual produto;
    private String avaliacao;

    public Avaliacao(long id, Date data, ProdutoAudioVisual produto, String avaliacao) {
        this.id = id;
        this.data = data;
        this.produto = produto;
        this.avaliacao = avaliacao;
    }
    public void alterarAvaliacao(String avaliacao) {
        this.avaliacao = avaliacao;
    }
    public void alterarProduto(ProdutoAudioVisual produto) {
        this.produto = produto;
    }

    public void imprimirAvaliacao() {
        System.out.println("==================================");
        produto.imprimir();
        System.out.println("----------------------------------");
        System.out.println("Avaliação (#" + id + "): " + avaliacao);
        System.out.println("Data: " + data);
        System.out.println("==================================\n");
    }
}
